using System;using System.IO;using System.Linq;using UnityEditor;using UnityEngine;using ZombieSurvival.Weapons;using ZombieSurvival.Inventory;using ZombieSurvival.Persistence;using AK74Reference;
public static class AK74FinalValidation {
 static int checks; static void Check(bool success,string description){if(!success)throw new Exception("AK74_ASSET_FAIL "+description);checks++;Debug.Log("AK74_ASSET_PASS "+description);}
 public static void Run(){var rifle=AssetDatabase.LoadAssetAtPath<ItemDefinition>(AK74GameplaySetup.ItemPath);var ammo=AssetDatabase.LoadAssetAtPath<ItemDefinition>(AK74GameplaySetup.AmmoPath);var weapon=rifle.Weapon;Check(weapon.Ammo==ammo&&weapon.MagazineSize==30&&weapon.automatic,"separate ammo magazine automatic");Check(weapon.handPose!=null&&weapon.bodyAnimations!=null&&weapon.bodyReloadSeconds>0,"explicit model hand/body profiles");var before=Directory.GetFiles("Assets/Weapons/AK74Gameplay","*",SearchOption.AllDirectories).Where(p=>!p.EndsWith(".meta")).ToDictionary(p=>p,p=>File.ReadAllBytes(p));AK74GameplaySetup.Run();Check(before.All(p=>p.Value.SequenceEqual(File.ReadAllBytes(p.Key))),"idempotent authoring preserves existing assets");
 foreach(var path in new[]{AK74GameplaySetup.Native,AssetDatabase.GetAssetPath(weapon.viewPrefab),AssetDatabase.GetAssetPath(rifle.WorldPrefab),AssetDatabase.GetAssetPath(ammo.WorldPrefab)}){var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(path);Check(prefab!=null&&prefab.GetComponentsInChildren<Transform>(true).All(t=>GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject)==0),"no missing scripts "+path);foreach(var r in prefab.GetComponentsInChildren<Renderer>(true)){Check(r.sharedMaterials.Length>0&&r.sharedMaterials.All(m=>m!=null),"materials "+r.name);}}
 var model=weapon.viewPrefab.GetComponentInChildren<AK74Presentation>(true);var material=model.magazineRenderers[0].sharedMaterial;Check(material.GetTexture("_BaseMap")!=null&&material.GetTexture("_BumpMap")!=null&&material.GetTexture("_MetallicGlossMap")!=null,"native PBR textures resolve after GUID repair");Check(weapon.viewPrefab.GetComponentInChildren<LODGroup>().lodCount==3,"three LOD share source rig");Check(weapon.attacks.Length==3&&weapon.attacks.All(a=>a!=null)&&Mathf.Abs(weapon.reloadSound.length-2.2f)<.01f,"attack and timed reload clips resolve");var player=AssetDatabase.LoadAssetAtPath<GameObject>(AK74GameplaySetup.Player);var data=new SerializedObject(player.GetComponent<PlayerSaveSystem>());var list=data.FindProperty("itemCatalog");foreach(var item in new[]{rifle,ammo})Check(Enumerable.Range(0,list.arraySize).Count(i=>list.GetArrayElementAtIndex(i).objectReferenceValue==item)==1,"single catalog entry "+item.Id);Check(player.GetComponent<PlayerWeapons>().FirearmHandPosition==new Vector3(.19f,-.045f,.03f),"authored pistol mount preserved");
 Debug.Log("AK74_ASSET_VALIDATION_OK checks="+checks);}
}
