    private static string LibraryHash()
    {
        const string folder="Assets/Environment/KursHub";
        var paths=Directory.GetFiles(folder,"*",SearchOption.AllDirectories).Where(p=>!p.EndsWith(".meta") && p.Replace('\\','/')!=folder+"/Models/KursHubShell.fbx" && !p.EndsWith("KursHub.prefab") && !p.EndsWith("KursHubManifest.json")).Select(p=>p.Replace('\\','/')).OrderBy(p=>p,StringComparer.Ordinal);
        using(var algorithm=System.Security.Cryptography.SHA256.Create())
        {
            Func<string,string> hash=p=>BitConverter.ToString(algorithm.ComputeHash(File.ReadAllBytes(p))).Replace("-","").ToLowerInvariant();
            var proof=string.Join("\n",paths.Select(p=>p+":"+hash(p)));
            return BitConverter.ToString(algorithm.ComputeHash(System.Text.Encoding.UTF8.GetBytes(proof))).Replace("-","").ToLowerInvariant();
        }
    }

    private IEnumerator Push(Vector3 direction,int frames)
    {
        transform.rotation=Quaternion.LookRotation(direction);
        for(int i=0;i<frames;i++){InputSystem.QueueStateEvent(keyboard,new KeyboardState(Key.W)); InputSystem.Update(); yield return null;}
        InputSystem.QueueStateEvent(keyboard,new KeyboardState());yield return null;
    }
    private IEnumerator Walk(Vector3 target,float expectedHeight,string label)
    {
        int count=0;
        while(Vector2.Distance(new Vector2(transform.position.x,transform.position.z),new Vector2(target.x,target.z)) > .16f && count++ < 2400)
        {
            var direction=target-transform.position;direction.y=0;transform.rotation=Quaternion.LookRotation(direction);
            InputSystem.QueueStateEvent(keyboard,new KeyboardState(Key.W)); InputSystem.Update(); yield return null;
            if(count==60) { var c=GetComponent<FirstPersonController>(); File.AppendAllText(Results,$"INPUT_DIAGNOSTIC enabled={c.enabled} accepts={c.AcceptingGameplayInput} action={c.FindGameplayAction("Move").ReadValue<Vector2>()} W={keyboard.wKey.isPressed} device={keyboard.enabled}\n"); }
            if(transform.position.y < -1)throw new Exception("Fell through floor: " + label + " " + transform.position);
        }
        InputSystem.QueueStateEvent(keyboard,new KeyboardState());
        for(int i=0;i<20;i++)yield return null;
        Require(count<2400 && Mathf.Abs(transform.position.y-expectedHeight)<.12f,
            label+" "+transform.position+" frames="+count);
    }
