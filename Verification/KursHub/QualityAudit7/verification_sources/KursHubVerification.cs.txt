    private static string CoverageSignature()
    {
        var registry=UnityEngine.Object.FindObjectsByType<ZombieSurvival.Persistence.WorldSaveRegistry>(FindObjectsInactive.Include).Single();
        var property=new SerializedObject(registry).FindProperty("sceneObjects");
        var authored=new System.Collections.Generic.HashSet<UnityEngine.Object>();
        for(int i=0;i<property.arraySize;i++)authored.Add(property.GetArrayElementAtIndex(i).objectReferenceValue);
        return string.Join("\n",UnityEngine.Object.FindObjectsByType<ZombieSurvival.Persistence.WorldSaveObject>(FindObjectsInactive.Include).Where(r=>r.gameObject.scene==registry.gameObject.scene && !authored.Contains(r)).Select(r=>r.name+"|"+r.Id+"|"+GlobalObjectId.GetGlobalObjectIdSlow(r)).OrderBy(v=>v,StringComparer.Ordinal));
    }
    public static string LibraryHash()
    {
        var paths=Directory.GetFiles(KursHubSetup.Folder,"*",SearchOption.AllDirectories).Where(p=>!p.EndsWith(".meta") && p.Replace('\\','/')!=KursHubSetup.Folder+"/Models/KursHubShell.fbx" && !p.EndsWith("KursHub.prefab") && !p.EndsWith("KursHubManifest.json")).Select(p=>p.Replace('\\','/')).OrderBy(p=>p,StringComparer.Ordinal);
        var text=string.Join("\n",paths.Select(p=>p.Replace('\\','/')+":"+Hash(p)));
        using(var algorithm=System.Security.Cryptography.SHA256.Create())
            return BitConverter.ToString(algorithm.ComputeHash(System.Text.Encoding.UTF8.GetBytes(text))).Replace("-","").ToLowerInvariant();
    }
    private static System.Collections.Generic.HashSet<string> EffectivePrefabIds(GameObject root)
    {
        var ids=new System.Collections.Generic.HashSet<string>();
        foreach(var transform in root.GetComponentsInChildren<Transform>(true))
        {
            var objects=new UnityEngine.Object[]{transform.gameObject}.Concat(transform.GetComponents<Component>().Where(c=>c!=null).Cast<UnityEngine.Object>());
            foreach(var obj in objects)
                if(AssetDatabase.TryGetGUIDAndLocalFileIdentifier(obj,out string guid,out long id))ids.Add(id.ToString(System.Globalization.CultureInfo.InvariantCulture));
        }
        File.WriteAllLines("../audit7/effective_prefab_ids.txt",ids.OrderBy(x=>x,StringComparer.Ordinal));
        return ids;
    }
    private static string PrefabIdentities()
    {
        var paths=Directory.GetFiles(KursHubSetup.Folder+"/Props/Prefabs","*.prefab")
            .Concat(Directory.GetFiles(KursHubSetup.Folder+"/InteriorMeshes","*.asset")).Append(KursHubSetup.PrefabPath).Select(p=>p.Replace('\\','/')).OrderBy(p=>p,StringComparer.Ordinal);
        return string.Join("\n",paths.Select(path=>path+"|"+AssetDatabase.AssetPathToGUID(path)+"|"+
            string.Join(",",Regex.Matches(File.ReadAllText(path),@"^--- !u!\d+ &(-?\d+)",RegexOptions.Multiline)
                .Cast<Match>().Select(m=>m.Groups[1].Value).OrderBy(id=>id,StringComparer.Ordinal))));
    }
    private static string Hash(string path)
    {
        using(var algorithm=System.Security.Cryptography.SHA256.Create())
            return BitConverter.ToString(algorithm.ComputeHash(File.ReadAllBytes(path))).Replace("-","").ToLowerInvariant();
    }
    public static void Require(bool value, string message) { if (!value) throw new Exception("KURS_TEST_FAILED: " + message); Pass(message); }
    public static void Pass(string message) { File.AppendAllText(Results, "PASS " + message + "\n"); Debug.Log("KURS_PROBE " + message); }
    private static void Preview(Camera camera, string name, Vector3 position, Vector3 target, float fov)
    {
        Directory.CreateDirectory("../audit7/previews");
        camera.transform.position = position; camera.transform.LookAt(target); camera.fieldOfView = fov;
        var rt = new RenderTexture(2400, 1500, 24, RenderTextureFormat.ARGB32);
        rt.Create();
        camera.targetTexture = rt;
        camera.Render();
        if (RenderPipeline.SupportsRenderRequest(camera, new UniversalRenderPipeline.SingleCameraRequest()))
            RenderPipeline.SubmitRenderRequest(camera, new UniversalRenderPipeline.SingleCameraRequest { destination = rt });
        var old = RenderTexture.active; RenderTexture.active = rt;
        var image = new Texture2D(2400,1500,TextureFormat.RGB24,false); image.ReadPixels(new Rect(0,0,2400,1500),0,0); image.Apply();
        File.WriteAllBytes("../audit7/previews/" + name + ".png", image.EncodeToPNG());
        RenderTexture.active = old; camera.targetTexture = null;
        UnityEngine.Object.DestroyImmediate(image); rt.Release(); UnityEngine.Object.DestroyImmediate(rt);
    }
