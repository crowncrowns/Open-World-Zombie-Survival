    public const string Results = "../stage3/play_validation.txt";
    public static void Require(bool condition,string text) { if(!condition)throw new Exception("KURS_TEST_FAILED: "+text); File.AppendAllText(Results,"PASS "+text+"\n"); Debug.Log("KURS_PROBE "+text); }
    private Keyboard keyboard;
    private CharacterController capsule;
    private void OnEnable() => Application.logMessageReceived += OnLog;
    private void OnDisable() => Application.logMessageReceived -= OnLog;
    private void OnLog(string condition, string stack, LogType type)
    {
        if (type != LogType.Exception && type != LogType.Error && type != LogType.Assert) return;
        if(stack.Contains("UnityEditor.Search.")) { File.AppendAllText(Results,"EDITOR_SEARCH_WARNING " + condition + "\n"); return; }
        File.AppendAllText(Results, "FAIL " + condition + "\n" + stack + "\n");
        EditorApplication.Exit(1);
    }
    private IEnumerator Start()
    {
        Time.captureFramerate = 60; QualitySettings.vSyncCount = 0;
        Application.runInBackground = true;
        InputSystem.settings.backgroundBehavior = InputSettings.BackgroundBehavior.IgnoreFocus;
        InputSystem.settings.editorInputBehaviorInPlayMode = InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
        keyboard = InputSystem.AddDevice<Keyboard>("KursProbeKeyboard"); capsule = GetComponent<CharacterController>();
        yield return null; yield return null;
        yield return Walk(new Vector3(-27,0,-98),0,"Actual lab south boundary opening and path");
        yield return Walk(new Vector3(0,0,-96),0,"Actual path turn");
        yield return Walk(new Vector3(0,0,-80),0,"Actual path to hub apron");

        yield return Walk(new Vector3(0,0,-61),0,"Vestibule centre");
        yield return Walk(new Vector3(3.8f,0,-61),0,"Vestibule east half");

        yield return Walk(new Vector3(2,0,-56),0,"Inner doorway east approach");

        yield return Walk(new Vector3(0,0,-62),0,"Return to entrance centre");
        yield return Walk(new Vector3(0,0,-44),0,"Main entrance from apron");

        yield return Walk(new Vector3(10,5.5f,20),5.5f,"East main staircase ascent");
        yield return Walk(new Vector3(-28,5.5f,25),5.5f,"Rear gallery crossing");
        yield return Walk(new Vector3(-39.5f,5.5f,28),5.5f,"West gallery");
        yield return Walk(new Vector3(-39.5f,5.5f,37.5f),5.5f,"Roof staircase approach");
        yield return Walk(new Vector3(-39.5f,8.25f,44),8.25f,"Roof stair first flight");
        yield return Walk(new Vector3(-36.5f,8.25f,44),8.25f,"Roof stair half-landing turn");
        yield return Walk(new Vector3(-36.5f,11,37.4f),11,"Roof stair second flight");
        yield return Walk(new Vector3(-36.5f,11,35),11,"Roof access doorway");
        yield return Walk(new Vector3(-30,11,33),11,"Walkable roof");
        yield return Walk(new Vector3(-23,11,29),11,"Roof route past shoulder blocks");
        yield return Walk(new Vector3(-23,11,-50),11,"Roof route to front cupola");
        yield return Walk(new Vector3(-14,11,-50),11,"Front cupola approach");
        yield return Walk(new Vector3(-23,11,-50),11,"Front cupola return");
        yield return Walk(new Vector3(-23,11,29),11,"Roof route return");
        yield return Walk(new Vector3(-30,11,33),11,"Roof return to access wing");
        yield return Walk(new Vector3(-36.5f,11,35),11,"Return to roof doorway");
        yield return Walk(new Vector3(-36.5f,11,37.4f),11,"Return to roof stair");
        yield return Walk(new Vector3(-36.5f,8.25f,44),8.25f,"Roof stair descent upper flight");

    private void Warp(Vector3 point)
    {
        InputSystem.QueueStateEvent(keyboard,new KeyboardState());
        capsule.enabled=false;transform.position=point;capsule.enabled=true;Physics.SyncTransforms();
    }
    private IEnumerator Push(Vector3 direction,int frames)
    {
        transform.rotation=Quaternion.LookRotation(direction);
        for(int i=0;i<frames;i++){InputSystem.QueueStateEvent(keyboard,new KeyboardState(Key.W)); InputSystem.Update(); yield return null;}
        InputSystem.QueueStateEvent(keyboard,new KeyboardState());yield return null;
    }
    private IEnumerator Walk(Vector3 target,float expectedHeight,string label)
    {
        int count=0;
        while(Vector2.Distance(new Vector2(transform.position.x,transform.position.z),new Vector2(target.x,target.z)) > .16f && count++ < 2400)
        {
            var direction=target-transform.position;direction.y=0;transform.rotation=Quaternion.LookRotation(direction);
            InputSystem.QueueStateEvent(keyboard,new KeyboardState(Key.W)); InputSystem.Update(); yield return null;
            if(count==60) { var c=GetComponent<FirstPersonController>(); File.AppendAllText(Results,$"INPUT_DIAGNOSTIC enabled={c.enabled} accepts={c.AcceptingGameplayInput} action={c.FindGameplayAction("Move").ReadValue<Vector2>()} W={keyboard.wKey.isPressed} device={keyboard.enabled}\n"); }
            if(transform.position.y < -1)throw new Exception("Fell through floor: " + label + " " + transform.position);
        }
        InputSystem.QueueStateEvent(keyboard,new KeyboardState());
        for(int i=0;i<20;i++)yield return null;
        Require(count<2400 && Mathf.Abs(transform.position.y-expectedHeight)<.12f,
            label+" "+transform.position+" frames="+count);
    }
}
