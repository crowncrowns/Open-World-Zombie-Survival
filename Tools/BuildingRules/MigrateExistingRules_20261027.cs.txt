            {
                var t=f.Find(face+bay);if(!t||!ZombieSurvival.ModularBuildings.Editor.BuildingGenerator.IsTripleWindow(t))throw new System.Exception("Triple missing "+b.name);

                if(t.GetComponentsInChildren<UnityEngine.Collider>(true).Length==0)throw new System.Exception("Triple colliders missing");
            }
        }
    }
    foreach(var b in entranceBuildings){var errors=ZombieSurvival.ModularBuildings.Editor.BuildingApartments.ValidateEntranceAccess(b);if(errors.Count>0)throw new System.Exception(string.Join(";",errors));}
