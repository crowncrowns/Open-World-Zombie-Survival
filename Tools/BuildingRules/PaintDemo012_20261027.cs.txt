bool RedField(int floor,int bay,bool rear)
{
    if(rear)bay=6-bay;
    if(floor<=2)return bay<=1;
    if(floor<=5)return bay<=2;
    if(floor<=8)return bay>=2&&bay<=4;
    return bay>=3&&bay<=5;
}

        material=RedField(floor,bay,module.name.StartsWith("Rear_"))?red:yellow;
    }
    else if(module.name=="LeftEnd"||module.name=="RightEnd")
    {
